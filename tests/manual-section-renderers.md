## Quick Test Script

```bash
# Generate all files
.venv/bin/python3 shared/generate_test_html.py --source cambridge-1 --all-skills --force

# Verify output
echo "=== File counts ==="
echo "Listening: $(ls .ielts/test-html/cambridge-1_listening_*.html 2>/dev/null | wc -l) files"
echo "Reading:   $(ls .ielts/test-html/cambridge-1_reading_*.html 2>/dev/null | wc -l) files"
echo "Speaking:  $(ls .ielts/test-html/cambridge-1_speaking_*.html 2>/dev/null | wc -l) files"
echo "Writing:   $(ls .ielts/test-html/cambridge-1_writing_*.html 2>/dev/null | wc -l) files"

# Verify no leftover template markers in any file
echo "=== Leftover markers ==="
grep -l '{{' .ielts/test-html/*.html 2>/dev/null || echo "None found — all clean"

# Verify all files have DOCTYPE
echo "=== Missing DOCTYPE ==="
for f in .ielts/test-html/*.html; do
  head -1 "$f" | grep -q '<!DOCTYPE' || echo "MISSING: $f"
done
echo "Done"

# Count total questions across all files
echo "=== Question counts ==="
python3 -c "
import json
idx = json.load(open('.ielts/test-html/_generated.json'))
for s in idx['sections']:
    print(f'  {s[\"skill\"]:12} test-{s[\"testNumber\"]} section-{s[\"sectionNumber\"]}: {s[\"questionCount\"]} questions')
print(f'Total: {sum(s[\"questionCount\"] for s in idx[\"sections\"])} questions across {len(idx[\"sections\"])} sections')
"
```
